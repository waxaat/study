def tokenize(text):
    separators = " \t\n.,;:!?()[]{}\"«»…—"
    tokens = []
    buffer = ""

    i = 0
    while i < len(text):
        ch = text[i]

        if ch == ".":
            prev_is_digit = i > 0 and text[i - 1].isdigit()
            next_is_digit = i + 1 < len(text) and text[i + 1].isdigit()
            if prev_is_digit and next_is_digit:
                buffer += ch
                i += 1
                continue

        if ch in separators:
            if buffer:
                tokens.append(buffer)
                buffer = ""
        else:
            buffer += ch

        i += 1

    if buffer:
        tokens.append(buffer)

    return tokens


print(tokenize("Привет, мир! Как дела?"))
print(tokenize("Жили-были 3.14 раза"))
print(tokenize('s = "don\'t"'))
print(tokenize("A...B"))
print(tokenize(""))

def window(lst, window_size):
    if window_size <= 0 or window_size > len(lst):
        return []
    return [tuple(lst[i:i + window_size]) for i in range(len(lst) - window_size + 1)]


print(window([1, 2, 3, 4, 5, 6, 7, 8, 9, 0], 3))

def replace_pair(text, marker, open_tag, close_tag):
    result = ""
    while True:
        start = text.find(marker)
        if start == -1:
            break
        end = text.find(marker, start + len(marker))
        if end == -1:
            break
        result += text[:start] + open_tag
        result += text[start + len(marker):end] + close_tag
        text = text[end + len(marker):]
    return result + text


def format_inline(text):
    text = replace_pair(text, "**", "<b>", "</b>")
    text = replace_pair(text, "`", "<code>", "</code>")
    return text


def markdown_to_html(md):
    lines = md.split("\n")
    result = []

    for line in lines:
        if line == "":
            result.append("")
            continue

        level = 0
        while level < len(line) and line[level] == "#":
            level += 1

        if 1 <= level <= 6 and line[level:level + 1] == " ":
            text = format_inline(line[level + 1:])
            result.append("<h" + str(level) + ">" + text + "</h" + str(level) + ">")
        else:
            result.append("<p>" + format_inline(line) + "</p>")

    return "\n".join(result)

assert markdown_to_html("# Привет") == "<h1>Привет</h1>"
assert markdown_to_html("### Тема") == "<h3>Тема</h3>"
assert markdown_to_html("###### Мелкий") == "<h6>Мелкий</h6>"
assert markdown_to_html("Это **важно**") == "<p>Это <b>важно</b></p>"
assert markdown_to_html("Код `x`") == "<p>Код <code>x</code></p>"
assert markdown_to_html("") == ""

md = "# Заголовок\n## Подзаголовок\nЭто **жирный** текст и `код`."
print(markdown_to_html(md))
