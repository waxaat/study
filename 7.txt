def func(x):
    return x**2 / 10 + 1

a, b = -2, 5

for N in [10, 100, 1000]:
    h = (b - a) / N
    f0 = func(a)
    f_last = func(b)
    S = (h / 2) * (f0 + f_last)
    for i in range(1, N - 1 + 1):
        x_i = a + i * h
        S += func(x_i) * h
    print(f"N={N}: {S:.2f}")

import random

n = int(input())

magic = [[0] * n for _ in range(n)]

i, j = 0, n // 2
for num in range(1, n * n + 1):
    magic[i][j] = num
    new_i, new_j = (i - 1) % n, (j + 1) % n
    if magic[new_i][new_j] != 0:
        new_i, new_j = (i + 1) % n, j
    i, j = new_i, new_j

# Перемешивание строк, столбцов и значений с сохранением свойства
rows = list(range(n))
random.shuffle(rows)
magic = [magic[r] for r in rows]

cols = list(range(n))
random.shuffle(cols)
magic = [[row[c] for c in cols] for row in magic]

if random.choice([True, False]):
    magic = [row[::-1] for row in magic[::-1]]  # разворот сохраняет свойство

for row in magic:
    print(*row)

const = n * (n * n + 1) // 2
print("Магическая константа:", const)

n = int(input())
treasures = []
for i in range(n):
    x, y = map(int, input().split())
    treasures.append([x, y])

ax, ay = map(int, input().split())

nearest = treasures[0]
min_dist = ((treasures[0][0] - ax)**2 + (treasures[0][1] - ay)**2) ** 0.5

for t in treasures:
    dist = ((t[0] - ax)**2 + (t[1] - ay)**2) ** 0.5
    if dist < min_dist:
        min_dist = dist
        nearest = t

print(nearest[0], nearest[1])
print(round(min_dist, 2))

def tokenize(text):
    separators = " \t\n.,;:!?()[]{}\"«»…—-"
    tokens = []
    buffer = ""

    i = 0
    while i < len(text):
        ch = text[i]

        if ch == ".":
            prev_is_digit = i > 0 and text[i-1].isdigit()
            next_is_digit = i + 1 < len(text) and text[i+1].isdigit()
            if prev_is_digit and next_is_digit:
                buffer += ch
                i += 1
                continue

        if ch in separators:
            if buffer:
                tokens.append(buffer)
                buffer = ""
        elif ch in "-'":
	    buffer += ch
        else:
            buffer += ch
        i += 1

    if buffer:
        tokens.append(buffer)

    return tokens


print(tokenize("Привет, мир! Как дела?"))
print(tokenize("Жили-были 3.14 раза"))
print(tokenize('s = "don\'t"'))
print(tokenize("A...B"))
print(tokenize(""))
